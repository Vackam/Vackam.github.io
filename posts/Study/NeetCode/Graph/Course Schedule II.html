선수 과목이 존재할 때 가능한 경우 중 하나를 구해라

푸는 방법

Course Schedule과 동일한데, result vector를 세우는 것 밖에 없다.

다음은 코드이다.

class Solution {
public:
    vector<int> findOrder(int n, vector<vector<int>>& p) {
        vector<vector<int>> adj(n);
        vector<int> in_d(n, 0);

        for (auto a : p) {
            adj[a[1]].emplace_back(a[0]);
            in_d[a[0]]++;
        } 

        queue<int> q;

        for (int i=0; i<n; ++i) {
            if (in_d[i] == 0) q.push(i);
        }

        vector<int> result;
        while (!q.empty()) {
            int target = q.front(); q.pop();
            result.push_back(target);

            for (auto a : adj[target]) {
                in_d[a]--;
                if (in_d[a] == 0) q.push(a);
            } 
        }

        if (result.size() != n) return {};
        return result;
    }
};
