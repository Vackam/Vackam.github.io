정직하게 하드코딩으로 풀자.

다음은 코드이다.

class Solution {
public:
    int calPoints(vector<string>& op) {
        stack<int> st;
        for (const auto& a : op) {
            if (a == "C") {
                st.pop();
            } else if (a == "D") {
                int target = st.top();
                st.push(target * 2);
            } else if (a == "+") {
                int first = st.top(); st.pop();
                int second = st.top();

                int target = first + second;

                st.push(first);
                st.push(target);
            } else {
                int target = stoi(a);
                st.push(target);
            }
        } 

        int result = 0;
        while (!st.empty()) {
            result += st.top(); st.pop();
        }

        return result;
    }
};
