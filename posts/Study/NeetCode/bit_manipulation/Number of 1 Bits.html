bit에서 1인 bit의 개수를 세시오

푸는 방법

정직하게 세면 된다.

다음은 코드이다

class Solution {
public:
    int hammingWeight(uint32_t n) {
        int counter = 0;
        while (n > 0) {
            if (n & 1) counter++;

            n = n >> 1;
        }

        return counter;
    }
};
