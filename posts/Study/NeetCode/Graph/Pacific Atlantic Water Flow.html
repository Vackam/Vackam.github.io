Pacific Atlantic Water Flow

2026-07-27 ps그래프

서론

| https://neetcode.io/problems/pacific-atlantic-water-flow/question

(위쪽, 왼쪽), (아래쪽, 오른쪽)을 전부 갈 수 있는 칸을 구하라.

푸는 방법

dp 테이블을 비트 연산을 통해 구하면 된다.

(위쪽, 왼쪽)을 1, (아래쪽, 오른쪽)을 2로 설정하고 초기에 접하는 칸들을 전부 큐에 넣는다. 이후 해당하는 칸부터 높이가 낮은 쪽에서 높은 쪽으로(역방향 탐색) 탐색을 진행하면서 높은 쪽 칸의 dp가 변경되면 해당 값을 넣어준다.

이후 마지막으로 dp를 돌면서 3인 값을 찾으면 정답이다.

다음은 코드이다.

class Solution {
public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        int m = heights.size();
        int n = heights[0].size();
        vector<vector<int>> dp(m, vector<int>(n, 0)); 
        queue<pair<int, int>> q;
        vector<vector<int>> result;

        vector<int> dx = {1, -1, 0, 0};
        vector<int> dy = {0, 0, 1, -1};
        // 1 - pacific, 2 - atlantic, 3 - both

        // pacific
        for (int j=0; j<n; ++j) {
            dp[0][j] |= 1;
            q.push({0, j});
        }
        for (int i=0; i<m; ++i) {
            dp[i][0] |= 1;
            q.push({i, 0});
        }

        // atlantic
        for (int j=0; j<n; ++j) {
            dp[m-1][j] |= 2;
            q.push({m-1, j});
        }
        for (int i=0; i<m; ++i) {
            dp[i][n-1] |= 2;
            q.push({i, n-1});
        }

        while (!q.empty()) {
            auto [x, y] = q.front(); q.pop();
            for (int i=0; i<4; ++i) {
                int nx = x + dx[i], ny = y + dy[i];

                if (nx < 0 || nx >= m || ny < 0 || ny >= n) continue;
                if (heights[nx][ny] < heights[x][y]) continue;
                int merged = dp[nx][ny] | dp[x][y];
                if (merged != dp[nx][ny]) {
                    dp[nx][ny] = merged;
                    q.push({nx, ny});
                }
            }
        }

        for (int i=0; i<m; ++i) {
            for (int j=0; j<n; ++j) {
                if (dp[i][j] == 3) result.push_back({i, j});
            }
        }

        return result;
    }
};

다른 좋은 풀이로는 단순하게 1인 애들, 2인 애들 각각 bfs 돌리고 서로 겹치는 칸만 결과로 내면 된다.
