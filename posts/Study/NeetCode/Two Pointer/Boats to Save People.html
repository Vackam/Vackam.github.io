푸는 방법

정렬한 후에 투 포인터로 탐색하면 된다.

다음은 코드이다.

class Solution {
public:
    int numRescueBoats(vector<int>& p, int limit) {
        // 정렬, 앞에서 뒤로 가야할 거 같은데
        int n = p.size();
        sort(p.begin(), p.end());

        int s = 0, e = n-1;

        int result = 0;
        while (s <= e) {
            if (p[s] + p[e] <= limit) {
                result++;
                s++;
                e--;
            } else {
                e--;
                result++;
            }
        }

        return result;
    }
};
