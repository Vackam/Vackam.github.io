| https://neetcode.io/problems/rotate-matrix/question

90도 회전한 행렬을 구하시오

푸는 방법

traspose 한 후 row 마다 reverse 해주면 된다.

다음은 코드이다

class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n = matrix.size();

        for (int i=0; i<n; ++i) {
            for (int j=i+1; j<n; ++j) {
                swap(matrix[i][j], matrix[j][i]);
            }
        }

        for (auto& row : matrix) reverse(row.begin(), row.end());
    }
};
