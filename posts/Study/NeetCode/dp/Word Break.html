dict를 통해 만들 수 있는 문자열인 지 판별하라

푸는 방법

dfs로 가능한 경우들을 map에 저장하고 끝낸 이후에 map에 해당 값이 있는 지 확인하면 된다.

다음은 코드이다.

class Solution {
public:
    bool is_in(const string& origin, const string& target) {
        int n = target.size();

        for (int i=0; i<n; ++i) {
            if (origin[i] != target[i]) return false;
        }

        return true;
    }
    void dfs(unordered_map<string, bool>& m, const vector<string>& w, const int& n, string target, string s) {
        string copy = target;

        if (!is_in(s, copy)) return;
        for (const auto& a : w) {
            copy += a;
            if (m.find(copy) == m.end() &&
                copy.size() <= n) {
                
                m[copy] = true;
                dfs(m, w, n, copy, s);
            }

            copy = target;
        }
    }
    bool wordBreak(string s, vector<string>& wordDict) {
        int n = s.size();

        unordered_map<string, bool> m;
        dfs(m, wordDict, n, "", s);

        return m.find(s) == m.end() ? false : true;
    }
};

본 함수 위 2개의 함수는 헬퍼 함수들이다.

하지만 모두가 알고 있듯이 문자열을 사용하면 시간이 느리다. 실제로 프루닝을 안하면 시간초과로 해결하지 못한다.

그래서 다음과 같은 풀이가 있다.

비슷한 방식을 생각하되 dp 처럼 생각하는 것이다.

여기까지 만들 수 있나? 그럼 오케이

이런 로직으로 지나간다.

다음은 코드이다.

class Solution {
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        int n = s.size();
        unordered_set<string> dict(wordDict.begin(), wordDict.end());
        vector<bool> dp(n+1, false);
        dp[0] = true;

        for (int i=1; i<=n; ++i) {
            for (int j=0; j<i; ++j) {
                if (dp[j] && dict.count(s.substr(j, i-j))) {
                    dp[i] = true;
                    break;
                }
            }
        }

        return dp[n];
    }
};
