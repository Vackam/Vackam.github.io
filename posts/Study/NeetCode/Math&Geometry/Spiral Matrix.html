class Solution {
public:
  vector<int> spiralOrder(vector<vector<int>>& matrix) {
    vector<int> result;

    int n = matrix.size(), m = matrix[0].size();

    int row_step = m, col_step = n-1;
    int x = 0, y = -1;
    int dir = 0;

    // 0 - 오른쪽
    // 1 - 아래쪽
    // 2 - 왼쪽
    // 3 - 위쪽

    while (result.size() < m * n) {
      switch (dir) {
        case 0:
          for (int i=y+1; i<=y+row_step; ++i) {
            result.push_back(matrix[x][i]);
          }
          y += row_step;
          row_step -= 1;
          dir = 1;
          break;
        case 1:
          for (int i=x+1; i<=x+col_step; ++i) {
            result.push_back(matrix[i][y]);
          }
          x += col_step;
          col_step -= 1;
          dir = 2;
          break;
        case 2:
          for (int i=y-1; i>=y-row_step; --i) {
            result.push_back(matrix[x][i]);
          }
          y -= row_step;
          row_step -= 1;
          dir = 3;
          break;
        case 3:
          for (int i=x-1; i>=x-col_step; --i) {
            result.push_back(matrix[i][y]);
          }
          x -= col_step;
          col_step -= 1;
          dir = 0;
          break;
      }
    }
    return result;
  }
};
