Design Hashset과 동일하게 풀었다.

다음은 코드이다.

class MyHashMap {
    vector<int> v;
public:
    MyHashMap() : v(1000001, -1) {}
    
    void put(int key, int value) {
        v[key] = value; 
    }
    
    int get(int key) {
        return v[key]; 
    }
    
    void remove(int key) {
        v[key] = -1; 
    }
};
