보물에 도착할 수 있는 최소 거리를 구하라

푸는 방법

성실하게 bfs로 탐색해 주면 된다.

뭔가 단순 덧셈으로 푸는 방식이 있어서 이상한 INF 값으로 막아둔 것 같은데, bfs로 성실하게 풀면 바로 풀린다.

다음은 코드이다.

constexpr int INF = 2147483647;

class Solution {
public:
    void islandsAndTreasure(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();

        vector<int> dx = {1, -1, 0, 0};
        vector<int> dy = {0, 0, 1, -1};

        queue<pair<int, int>> q;

        for (int i=0; i<m; ++i) {
            for (int j=0; j<n; ++j) {
                if (grid[i][j] == 0) {
                    q.push({i, j});
                }
            }
        }

        int kan = 0;
        while(!q.empty()) {
            kan++;
            int size = q.size();

            for (int i=0; i<size; ++i) {
                auto a = q.front();
                q.pop();

                for (int j=0; j<4; ++j) {
                    int nx = a.first + dx[j];
                    int ny = a.second + dy[j];

                    if (
                        nx >= 0 && nx < m &&
                        ny >= 0 && ny < n &&
                        grid[nx][ny] == INF
                    ) {
                        q.push({nx, ny});
                        grid[nx][ny] = kan;
                    }
                }                
            } 
        }

        return;
    }
};
