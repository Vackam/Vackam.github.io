Minimum Size Subarray Sum

target보다 큰 값을 가지는 부분 수열 중 가장 작은 길이를 반환하라

푸는 방법

누적합을 구하고, 이후 two pointer 방식으로 풀면 된다.

다음은 코드이다.

class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int n = nums.size();

        vector<int> gum(n+1, 0);

        for (int i=1; i<=n; ++i) {
            gum[i] = nums[i-1] + gum[i-1];
        }

        int s = 0, e = 0;
        int result = 1000000;
        while (e <= n) {
            int k = gum[e] - gum[s];

            if (k < target) {
                e++;
            } else if (k >= target) {
                result = min(result, e-s);
                s++;
            } 
        }

        return result == 1000000 ? 0 : result;
    }
};
