Min Cost to Connect Points

2026-09-02 psAdvanced-Graph

서론

| https://neetcode.io/problems/min-cost-to-connect-points/question

모든 점을 갈 수 있는 최소 거리를 구하라.

푸는 방법

mst를 구하는 문제이다.

크루스칼 방식으로 풀었고, 이를 위해 union find를 사용했다.

다음은 코드이다.

using pipii = pair<int, pair<int, int>>;
class Solution {
public:
    vector<int> mother;
    pipii make_dis(const vector<int>& a, const vector<int>& b, int i, int j) {
        return {
            abs(a[0] - b[0]) + abs(a[1] - b[1]),
            {
                i, j 
            }
        };
    }

    int find(int x) {
        if (x == mother[x]) return x;
        return mother[x] = find(mother[x]);
    }
    
    void merge(int u, int v) {
        u = find(u);
        v = find(v);

        if (u == v) {
            return;
        }
        mother[u] = v;
    }

    int minCostConnectPoints(vector<vector<int>>& points) {
        // 이거 mst 만드는 문제 같은데
        // O(n^2) 으로 거리 다 만든 다음에
        // 젤 작은 거 부터 계속 만들면 될 거 같은데
        // 그리고 머지 때리면서 union find 하면 답 나오지 않을까?
        int n = points.size();
        mother.clear();
        mother.resize(n);
        iota(mother.begin(), mother.end(), 0);

        priority_queue<pipii, vector<pipii>, greater<pipii>> q;

        for (int i=0; i<n; ++i) {
            for (int j=i+1; j<n; ++j) {
                q.push(make_dis(points[i], points[j], i, j));
            }
        }

        int result = 0;
        int counter = 0;
        while (!q.empty()) {
            if (counter == n-1) break;
            auto a = q.top();
            q.pop();

            auto [u, v] = a.second;

            if (find(u) != find(v)) {
                result += a.first;
                merge(u, v);
                counter++;
            }
        }

        return result;
    }
};
