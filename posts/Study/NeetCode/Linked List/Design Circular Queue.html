벡터를 하나 세운 다음에

  1. 적재를 얼마나 했는가
  2. 시작점이 어디인가
  3. 끝점이 어디인가

를 체크해가면서 나가면 아주 단순하게 풀 수 있다.

다음은 코드이다.

class MyCircularQueue {
public:
    vector<int> q;
    int start, end, count, k;
    MyCircularQueue(int k) : q(k), start(0), end(k-1), count(0), k(k) {}
    
    bool enQueue(int value) {
        if (isFull()) return false;
        end = (end + 1) % k;
        q[end] = value;
        count++;
        return true;
    }
    
    bool deQueue() {
        if (isEmpty()) return false;
        start = (start + 1) % k;
        count--;
        return true;    
    }
    
    int Front() {
        return isEmpty() ? -1 : q[start]; 
    }
    
    int Rear() {
        return isEmpty() ? -1 : q[end]; 
    }
    
    bool isEmpty() {
        return count == 0; 
    }
    
    bool isFull() {
        return count == k; 
    }
};
