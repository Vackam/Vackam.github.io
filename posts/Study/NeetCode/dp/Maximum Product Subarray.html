Maximum Product Subarray

2026-08-08 psdp

서론

| https://neetcode.io/problems/maximum-product-subarray/question

곱이 가장 큰 부분배열을 찾아라

푸는 방법

핵심은 0과 음수를 어떻게 처리하냐이다. 저것들만 없으면 무조건 풀 배열이 정답이다.

첫번째 풀이는 상한과 하한을 들고 다니는 방법이다. 음수로 생긴 값도 들고 다니면서 다시 나올 음수를 기다리는 느낌이다.

다음은 코드이다

class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int n = nums.size();

        int cur_max = nums[0], cur_min = nums[0], ans = nums[0];

        for (int i=1; i<n; ++i) {
            int a = nums[i];
            if (a < 0) swap(cur_max, cur_min);
            cur_max = max(a, cur_max*a);
            cur_min = min(a, cur_min * a);
            ans = max(ans, cur_max);
        }
 
        return ans;
    }
};

다른 풀이로는 0을 기준점으로 앞에서 탐색하고 뒤에서 탐색하는 방식이다.

다음은 코드이다.

class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int n = nums.size(), ans = INT_MIN;
        long long pre = 1, suf = 1;

        for (int i=0; i<n; ++i) {
            pre *= nums[i];
            suf *= nums[n-1-i];
            ans = max(ans, (int)max(pre,suf));
            if (pre == 0) pre = 1;
            if (suf == 0) suf = 1;
        }    

        return ans;
    }
};
