Longest Increasing Path in Matrix

2026-09-12 psdp

서론

| https://neetcode.io/problems/longest-increasing-path-in-matrix/question

가장 긴 증가하는 path를 찾으시오

푸는 방법

큰 값부터 하나씩 더해가면서 dp로 풀었다.

다음은 코드이다.

using pii = pair<int, int>;

class Solution {
public:
    int longestIncreasingPath(vector<vector<int>>& matrix) {
        int n = matrix.size(), m = matrix[0].size();
        int dx[] = {1, -1, 0, 0};
        int dy[] = {0, 0, 1, -1};

        priority_queue<pair<int, pii>> q;

        for (int i=0; i<n; ++i) {
            for (int j=0; j<m; ++j) {
                q.push({ matrix[i][j], {i, j} });
            }
        } 

        vector<vector<int>> v(n, vector<int>(m, 1));

        while (!q.empty()) {
            auto a = q.top(); q.pop();
            auto [x, y] = a.second;

            for (int i=0; i<4; ++i) {
                int nx = x + dx[i];
                int ny = y + dy[i];

                if (nx < 0 || nx >= n || ny < 0 || ny >= m || (a.first) <= matrix[nx][ny]) continue;

                v[nx][ny] = max(v[nx][ny], v[x][y] + 1);
            }
        }

        int result = 0;
        for (int i=0; i<n; ++i) {
            for (int j=0; j<m; ++j) {
                result = max(result, v[i][j]);
            }
        } 

        return result;
    }
};

순회하면서 큐에 값을 집어넣은 다음, 큰 값에서 계속 내려오도록 하였다.

path는 증가하는 path이지만, 문제가 작은 값을 하게 되면 생길 거 같아서 맘 편하게 큰값에서 내려왔다.

다만 우선순위 큐를 씀으로 최적이 아니게 되었다. 최적 코드는 다음과 같다.

class Solution {
public:
    int n, m;
    int dx[4] = {1,-1,0,0};
    int dy[4] = {0,0,1,-1};
    
    int dfs(vector<vector<int>>& matrix, vector<vector<int>>& memo, int x, int y) {
        if (memo[x][y]) return memo[x][y];
        memo[x][y] = 1;
        for (int i = 0; i < 4; ++i) {
            int nx = x+dx[i], ny = y+dy[i];
            if (nx<0||nx>=n||ny<0||ny>=m||matrix[nx][ny]<=matrix[x][y]) continue;
            memo[x][y] = max(memo[x][y], dfs(matrix, memo, nx, ny) + 1);
        }
        return memo[x][y];
    }

    int longestIncreasingPath(vector<vector<int>>& matrix) {
        n = matrix.size(); m = matrix[0].size();
        vector<vector<int>> memo(n, vector<int>(m, 0));
        int res = 0;
        for (int i=0; i<n; ++i)
            for (int j=0; j<m; ++j)
                res = max(res, dfs(matrix, memo, i, j));
        return res;
    }
};

dp를 채우면서 dfs를 도는 것이다.
