class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        int n = digits.size();

        int upper = 1;
        for (int i=n-1; i>=0; --i) {
            auto& target = digits[i];

            if (upper + target == 10) {
                upper = 1;
                target = 0;
            } else {
                target += upper;
                upper = 0;
            }
        }

        if (upper == 1) {
            digits.insert(digits.begin(), 1);
        }

        return digits;
    }
};
