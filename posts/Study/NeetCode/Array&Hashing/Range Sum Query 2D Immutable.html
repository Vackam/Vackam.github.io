Range Sum Query 2D Immutable

주어진 사각형의 합을 구하라

푸는 방법

sumRegion이 들어오면 계산을 하면 되겠지만, 이 문제에서는 O(1)을 요구한다. 그러면 누적합으로 풀면 된다.

다음은 코드이다.

class NumMatrix {
    vector<vector<int>> numeric_sum;
public:
    NumMatrix(vector<vector<int>>& matrix) {
        // 누적합 만들어서 하는 거 가틍ㄴ데

        int m = matrix.size(), n = matrix[0].size();
        numeric_sum.assign(m+1, vector<int>(n+1, 0));

        for (int i=0; i<m; ++i) {
            for (int j=0; j<n; ++j) {
                numeric_sum[i+1][j+1] = matrix[i][j] + numeric_sum[i][j+1] + numeric_sum[i+1][j] - numeric_sum[i][j];
            }
        }
    }
    
    int sumRegion(int row1, int col1, int row2, int col2) {
        return numeric_sum[row2+1][col2+1] - numeric_sum[row1][col2+1] - numeric_sum[row2+1][col1] + numeric_sum[row1][col1]; 
    }
};
