회의를 하기 위한 최소 개수를 구하시오

푸는 방법

미팅룸을 하나 넣고, 만약 겹친다면 새롭게 방을 추가하면 된다.

이 로직은 min-heap으로 만들면 된다.

다음은 코드이다.

class Solution {
public:
    int minMeetingRooms(vector<Interval>& intervals) {
        int n = intervals.size();
        if (n < 2) return n;

        priority_queue<int, vector<int>, greater<int>> q;

        sort(intervals.begin(), intervals.end(),
            [](const Interval& a, const Interval& b) {
                return tie(a.start, a.end) < tie(b.start, b.end);
        });

        q.push(intervals[0].end);
        for (int i=1; i<n; ++i) {
            const auto& a = intervals[i];
            if (q.top() <= a.start) {
                q.pop();
            }
            q.push(a.end);
        }
        return q.size();
    }
};
