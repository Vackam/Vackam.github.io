Reverse Bits

bit를 거울 반전 시키시오

푸는 방법

마스킹 밀고 앤드 연산 하기

이걸 정직하게 하면 된다.

class Solution {
public:
    uint32_t reverseBits(uint32_t n) {
        uint32_t result = 0;
        for (int i=0; i<32; ++i) {
            result = (result << 1) | (n & 1);
            n >>= 1;
        }
        
        return result;
    }
};
