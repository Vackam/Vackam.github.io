Detect Squares

| https://neetcode.io/problems/count-squares/question

만들 수 있는 정사각형의 개수를 세라

푸는 방법

1000이 상한선 이므로, 1000 * 1000 이차원 배열을 구성한다. 이후 점을 축적하고, count 시에는 각 점의 개수를 곱함으로 가능한 정사각형의 개수를 구할 수 있다.

다음은 코드이다.

class CountSquares {
    vector<vector<int>> cnt;
public:
    CountSquares() : cnt(1001, vector<int>(1001, 0)) {
    }
    
    void add(vector<int> point) {
        cnt[point[0]][point[1]]++;
    }
    
    int count(vector<int> point) {
        int x = point[0], y = point[1], res = 0;

        for (int y2 = 0; y2 <= 1000; y2++) {
            if (y2 == y || cnt[x][y2] == 0) continue;

            int d = abs(y2 - y);

            if (x + d <= 1000) 
                res += cnt[x][y2] * cnt[x+d][y] * cnt[x+d][y2];
            if (x - d >= 0) 
                res += cnt[x][y2] * cnt[x-d][y] * cnt[x-d][y2];
        }

        return res;
    }
};
