Non-Cyclical Number

각 자릿수의 제곱합이 결국 1이 되는 지를 확인하라

푸는 방법

visited 배열을 세워서 풀면 아주 쉽다.

다음은 코드이다.

class Solution {
public:
    bool isHappy(int n) {
        vector<bool> visited(1000 + 1, false);

        while (!visited[n]) {
            visited[n] = true;
            int temp = 0;

            while (n > 0) {
                int a = n % 10;
                temp += a * a;
                n /= 10;
            }

            n = temp;
        }    

        return visited[1];
    }
};
