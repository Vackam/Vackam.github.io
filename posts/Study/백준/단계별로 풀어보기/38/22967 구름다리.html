22967 구름다리

2026-11-28 ps단계별로_풀어보기백준

서론

| https://acmicpc.net/problem/22967

구름 다리를 추가로 놓아서 지름을 최소화 시켜보자.

푸는 방법

이상적인 방향인 지름이 1인 경우가 가능한 지 생각해보자. 그런 경우는 어떤 점이 다른 모든 점과 연결되어 있어야 한다. 서로 다른 정점 n에 대해서, 다른 두 정점이 연결되는 경우의 수는 \(n \choose 2\)이다. 문제 조건에서 이미 n-1개의 간선이 존재하고 우리가 추가로 그을 수 있는 간선은 n-1개이다. 즉, \(2(n-1) \ge {n \choose 2} = {n(n-1) \div 2}\) 를 만족하는 n에 대해 1인 경우가 존재한다. 결과를 보면 n <= 4인 경우에만 1이 가능하다.

그럼 지름이 1인 경우는 제한적이므로 구했다. 그럼 n이 5 이상인 경우는 어떻게 되는가?

이 경우는 단순한데, 전부 지름이 2인 경우로 만들면 된다. 아무 정점을 하나 정하고 이 정점을 다른 모든 정점과 연결시키면 된다. 그렇게 되면 어느 점에서 시작하든 최대 2번의 간선을 거쳐서 다른 모든 정점으로 갈 수 있다. 또한, 이렇게 했을 때 추가되는 간선이 정확히 n-1개 이므로 이상적인 정답은 아니지만 확실한 정답이다.

다음은 코드이다.

#include <bits/stdc++.h>

using namespace std;

constexpr int MAX_VAL = 201;

int main() {
  int n; cin >> n;
  
  vector<vector<int>> v(n + 1, vector<int>(n + 1, MAX_VAL));

  if (n == 2) {
    int trash1, trash2; cin >> trash1 >> trash2;
    cout << "0\n1";
  } else if (n <= 4) {
    // 지름 1이 가능
    
    vector<pair<int, int>> result;
    for (int i=1; i<n; ++i) {
      int a, b; cin >> a >> b;

      v[a][b] = 1;
      v[b][a] = 1;
    } 

    for (int i=1; i<=n; ++i) {
      for (int j=i+1; j<=n; ++j) {
        if (v[i][j] == MAX_VAL) result.push_back({i, j}); 
      }
    }

    cout << result.size() << "\n";
    cout << 1 << "\n";
    for (auto a : result) {
      cout << a.first << " " << a.second << "\n";
    }
  } else {
    vector<pair<int, int>> result;
    for (int i=1; i<n; ++i) {
      int a, b; cin >> a >> b;
      v[min(a, b)][max(a, b)] = 1;
    }

    // 1 로 고정
    for (int i=2; i<=n; ++i) {
      if (v[1][i] == MAX_VAL) {
        result.push_back({1, i});
      }
    } 

    cout << result.size() << "\n";
    cout << 2 << "\n";
    for (auto a : result) {
      cout << a.first << " " << a.second << "\n";
    }
  }

  return 0;
}
