점프를 몇 번 뛰어야 하는 지 구하라

푸는 방법

이번에도 동일하게 bfs를 생각했다.

아래는 코드이다.

class Solution {
public:
    int jump(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return 0;

        queue<int> q;
        vector<bool> visited(n, false);

        q.push(0);
        visited[0] = true;
        int jump = 0;

        while (!q.empty()) {
            jump++;
            int turn = q.size();

            for (int i=0; i<turn; ++i) {
                auto a = q.front();
                q.pop();

                for (int j=1; j<=nums[a]; ++j) {
                    if (a + j >= n - 1) return jump;
                    if (!visited[a + j]) {
                        q.push(a + j);
                        visited[a + j] = true;
                    }
                }
            }
        }

        return 0;
    }
};

이것보다 더 단순한 버전이 있다.

그리디로 구간을 넓힌다고 생각하면 된다. 위에서는 각 갈 수 있는 칸을 queue로 저장해서 풀었으면, 여기서는 가능성을 넓혀 놓는 것이다.

한 칸에서 갈 수 있는 범위를 curEnd로 하고 curEnd에 속하는 칸들은 이미 접근이 가능한 범위니까 무시한다. 이후 범위가 벗어나면 그때 curEnd를 수정해나가면서 진행한다.

class Solution {
public:
    int jump(vector<int>& nums) {
        int n = nums.size();
        int jumps = 0, curEnd = 0, farthest = 0;
        for (int i = 0; i < n - 1; ++i) {  // 마지막 칸은 볼 필요 없음
            farthest = max(farthest, i + nums[i]);
            if (i == curEnd) {      // 현재 점프 범위의 끝에 도달
                jumps++;
                curEnd = farthest;  // 다음 점프 범위로 확장
            }
        }
        return jumps;
    }
};
